name: Scheduled data refresh

# Keeps the app current without needing Vercel Pro. Every 2 hours it refreshes the macro
# briefing (rates calendar + market-moving headlines) and one rotating research industry;
# once a day it also triggers the full analysis refresh + collect (a backup to the Vercel crons).
#
# Setup: add two repo secrets (Settings → Secrets and variables → Actions):
#   APP_URL      = https://your-production-domain            (no trailing slash)
#   CRON_SECRET  = the same value set as CRON_SECRET in Vercel
# You can also run it on demand from the Actions tab ("Run workflow").

on:
  schedule:
    - cron: "0 */2 * * *" # every 2 hours — macro + rotating research
    - cron: "15 2 * * *" # daily 02:15 UTC — full analysis refresh + collect
  workflow_dispatch: {}

concurrency:
  group: scheduled-refresh
  cancel-in-progress: false

jobs:
  macro:
    if: github.event.schedule == '0 */2 * * *' || github.event_name == 'workflow_dispatch'
    runs-on: ubuntu-latest
    steps:
      - name: Refresh macro + research
        env:
          APP_URL: ${{ secrets.APP_URL }}
          CRON_SECRET: ${{ secrets.CRON_SECRET }}
        run: |
          if [ -z "$APP_URL" ] || [ -z "$CRON_SECRET" ]; then
            echo "APP_URL and CRON_SECRET repo secrets are required."; exit 1
          fi
          curl -fsS --max-time 280 "$APP_URL/api/cron/macro" -H "Authorization: Bearer $CRON_SECRET"

  daily:
    if: github.event.schedule == '15 2 * * *'
    runs-on: ubuntu-latest
    steps:
      - name: Nightly refresh + collect
        env:
          APP_URL: ${{ secrets.APP_URL }}
          CRON_SECRET: ${{ secrets.CRON_SECRET }}
        run: |
          if [ -z "$APP_URL" ] || [ -z "$CRON_SECRET" ]; then
            echo "APP_URL and CRON_SECRET repo secrets are required."; exit 1
          fi
          curl -fsS --max-time 290 "$APP_URL/api/cron/refresh" -H "Authorization: Bearer $CRON_SECRET"
          curl -fsS --max-time 290 "$APP_URL/api/cron/collect" -H "Authorization: Bearer $CRON_SECRET"
